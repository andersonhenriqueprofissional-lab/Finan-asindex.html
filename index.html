// ========================================== // 1. GERENCIADOR DE ESTADO & INICIALIZAÇÃO // ========================================== // Inicializa a aplicação ao carregar a página document.addEventListener('DOMContentLoaded', () => { inicializarGerenciadorEstado(); inicializarFiltroCompetencia(); inicializarSistemaAbas(); }); function inicializarGerenciadorEstado() { let lancamentos = JSON.parse(localStorage.getItem('lancamentos')) || []; // Ordena os lançamentos por data (mais recentes primeiro) lancamentos.sort((a, b) => new Date(b.data) - new Date(a.data)); localStorage.setItem('lancamentos', JSON.stringify(lancamentos)); } // ========================================== // 2. DASHBOARD DE RESUMOS // ========================================== const totalRecebidoCard = document.getElementById('totalRecebido'); const totalContasCard = document.getElementById('totalContas'); const totalMetasCard = document.getElementById('totalMetas'); const saldoRestanteValorSpan = document.getElementById('saldoRestanteValor'); function formatarMoeda(valor) { return valor.toLocaleString('pt-BR', { style: 'currency', currency: 'BRL' }); } // Atualiza os cards com base nos dados que estão atualmente renderizados/filtrados function atualizarResumo(lancamentosVisiveis) { let totalRecebido = 0; let totalContas = 0; let totalMetas = 0; lancamentosVisiveis.forEach(l => { const valor = parseFloat(l.valor) || 0; if (l.tipo.startsWith('receita')) { totalRecebido += valor; } else if (l.tipo.startsWith('pagamento')) { totalContas += valor; } else if (l.tipo === 'meta' || l.tipo === 'investimento') { totalMetas += valor; } }); if (totalRecebidoCard) totalRecebidoCard.innerText = formatarMoeda(totalRecebido); if (totalContasCard) totalContasCard.innerText = formatarMoeda(totalContas); if (totalMetasCard) totalMetasCard.innerText = formatarMoeda(totalMetas); const saldoLiquido = totalRecebido - totalContas - totalMetas; if (saldoRestanteValorSpan) { saldoRestanteValorSpan.innerText = formatarMoeda(saldoLiquido); saldoRestanteValorSpan.style.color = saldoLiquido >= 0 ? 'green' : 'red'; } } // ========================================== // 3. FORMULÁRIO E SUBMISSÃO DE LANÇAMENTOS // ========================================== const formLancamento = document.getElementById('formLancamento'); if (formLancamento) { formLancamento.addEventListener('submit', function(event) { event.preventDefault(); const editingId = parseInt(this.dataset.editingId); if (!isNaN(editingId)) { salvarEdicaoLancamento(editingId); } else { const inputDescricao = document.getElementById('descricaoLancamento'); const inputValor = document.getElementById('valorLancamento'); const selectTipo = document.getElementById('tipoLancamento'); const inputData = document.getElementById('dataLancamento'); const inputCategoria = document.getElementById('categoriaLancamento'); const novoLancamento = { id: Date.now(), descricao: inputDescricao.value, valor: parseFloat(inputValor.value.replace(',', '.')), tipo: selectTipo.value, data: inputData.value, categoria: inputCategoria.value, status: 'pendente' }; adicionarLancamento(novoLancamento); this.reset(); inputDescricao.focus(); } }); } function adicionarLancamento(lancamento) { let lancamentos = JSON.parse(localStorage.getItem('lancamentos')) || []; lancamentos.push(lancamento); localStorage.setItem('lancamentos', JSON.stringify(lancamentos)); // Força a atualização da tabela respeitando o filtro atual const inputCompetencia = document.getElementById('filtroCompetencia'); if (inputCompetencia && inputCompetencia.value) { filtrarLancamentosPorCompetencia(inputCompetencia.value); } else { renderizarTabelaFiltrada(lancamentos); } } // ========================================== // 4. RENDERIZAÇÃO DA TABELA E STATUS // ========================================== const tabelaLancamentosBody = document.getElementById('tabelaLancamentosBody'); function renderizarTabelaFiltrada(lancamentosParaRenderizar) { if (!tabelaLancamentosBody) return; tabelaLancamentosBody.innerHTML = ''; // Ordena por data (mais recentes primeiro) lancamentosParaRenderizar.sort((a, b) => new Date(b.data) - new Date(a.data)); lancamentosParaRenderizar.forEach(lancamento => { const linha = tabelaLancamentosBody.insertRow(); linha.dataset.lancamentoId = lancamento.id; const dataCell = linha.insertCell(); dataCell.textContent = new Date(lancamento.data).toLocaleDateString('pt-BR', {timeZone: 'UTC'}); const descricaoCell = linha.insertCell(); descricaoCell.textContent = lancamento.descricao; const categoriaCell = linha.insertCell(); categoriaCell.textContent = lancamento.categoria || '-'; const tipoCell = linha.insertCell(); tipoCell.textContent = lancamento.tipo.replace('_', ' '); const valorCell = linha.insertCell(); const valorFormatado = parseFloat(lancamento.valor).toLocaleString('pt-BR', { style: 'currency', currency: 'BRL' }); valorCell.textContent = valorFormatado; if (lancamento.tipo.startsWith('pagamento') || lancamento.tipo === 'meta' || lancamento.tipo === 'investimento') { valorCell.style.color = 'red'; } else { valorCell.style.color = 'green'; } // Renderiza o botão de status renderizarBotoesStatus(lancamento, linha); // Ações const acoesCell = linha.insertCell(); acoesCell.innerHTML = ` `; }); // Atualiza o dashboard com os dados que estão visíveis na tela atualizarResumo(lancamentosParaRenderizar); } function renderizarBotoesStatus(lancamento, linha) { const statusCell = linha.insertCell(); const statusButton = document.createElement('button'); statusButton.className = `status-btn`; switch (lancamento.status) { case 'concluido': statusButton.textContent = 'Concluído'; statusButton.classList.add('🟢'); break; case 'atrasado': statusButton.textContent = 'Atrasado'; statusButton.classList.add('🔴'); break; case 'pendente': default: statusButton.textContent = 'Pendente'; statusButton.classList.add('🟡'); lancamento.status = 'pendente'; break; } statusButton.onclick = (event) => { event.stopPropagation(); alterarStatusCiclo(event, lancamento.id); }; statusCell.appendChild(statusButton); } function alterarStatusCiclo(event, lancamentoId) { const button = event.target; const statusAtual = button.textContent; let novoStatus = ''; let novoStatusButtonText = ''; let novoStatusClass = ''; switch (statusAtual) { case 'Pendente': novoStatus = 'atrasado'; novoStatusButtonText = 'Atrasado'; novoStatusClass = '🔴'; break; case 'Atrasado': novoStatus = 'concluido'; novoStatusButtonText = 'Concluído'; novoStatusClass = '🟢'; break; case 'Concluído': default: novoStatus = 'pendente'; novoStatusButtonText = 'Pendente'; novoStatusClass = '🟡'; break; } button.textContent = novoStatusButtonText; button.classList.remove('🟢', '🟡', '🔴'); button.classList.add(novoStatusClass); let lancamentos = JSON.parse(localStorage.getItem('lancamentos')) || []; const lancamentoIndex = lancamentos.findIndex(l => l.id === lancamentoId); if (lancamentoIndex !== -1) { lancamentos[lancamentoIndex].status = novoStatus; localStorage.setItem('lancamentos', JSON.stringify(lancamentos)); // Recarrega os dados atuais para atualizar o dashboard const inputCompetencia = document.getElementById('filtroCompetencia'); if (inputCompetencia && inputCompetencia.value) { filtrarLancamentosPorCompetencia(inputCompetencia.value); } else { atualizarResumo(lancamentos); } } } // ========================================== // 5. AÇÕES (EDITAR, SALVAR, EXCLUIR, DUPLICAR) // ========================================== function editarLancamento(id) { let lancamentos = JSON.parse(localStorage.getItem('lancamentos')) || []; const lancamentoParaEditar = lancamentos.find(l => l.id === id); if (lancamentoParaEditar) { document.getElementById('descricaoLancamento').value = lancamentoParaEditar.descricao; document.getElementById('valorLancamento').value = lancamentoParaEditar.valor.toFixed(2).replace('.', ','); document.getElementById('tipoLancamento').value = lancamentoParaEditar.tipo; document.getElementById('dataLancamento').value = lancamentoParaEditar.data; document.getElementById('categoriaLancamento').value = lancamentoParaEditar.categoria || ''; const form = document.getElementById('formLancamento'); form.dataset.editingId = id; const submitButton = form.querySelector('button[type="submit"]'); if (submitButton) submitButton.textContent = 'Atualizar Lançamento'; } else { alert('Lançamento não encontrado para edição.'); } } function salvarEdicaoLancamento(id) { const form = document.getElementById('formLancamento'); let lancamentos = JSON.parse(localStorage.getItem('lancamentos')) || []; const indiceLancamento = lancamentos.findIndex(l => l.id === id); if (indiceLancamento !== -1) { lancamentos[indiceLancamento] = { ...lancamentos[indiceLancamento], descricao: document.getElementById('descricaoLancamento').value, valor: parseFloat(document.getElementById('valorLancamento').value.replace(',', '.')), tipo: document.getElementById('tipoLancamento').value, data: document.getElementById('dataLancamento').value, categoria: document.getElementById('categoriaLancamento').value }; localStorage.setItem('lancamentos', JSON.stringify(lancamentos)); form.dataset.editingId = ''; form.reset(); const submitButton = form.querySelector('button[type="submit"]'); if (submitButton) submitButton.textContent = 'Adicionar Lançamento'; const inputCompetencia = document.getElementById('filtroCompetencia'); if (inputCompetencia && inputCompetencia.value) { filtrarLancamentosPorCompetencia(inputCompetencia.value); } else { renderizarTabelaFiltrada(lancamentos); } } } function excluirLancamento(id) { if (!confirm('Tem certeza que deseja excluir este lançamento?')) return; let lancamentos = JSON.parse(localStorage.getItem('lancamentos')) || []; const lancamentosAtualizados = lancamentos.filter(l => l.id !== id); localStorage.setItem('lancamentos', JSON.stringify(lancamentosAtualizados)); const inputCompetencia = document.getElementById('filtroCompetencia'); if (inputCompetencia && inputCompetencia.value) { filtrarLancamentosPorCompetencia(inputCompetencia.value); } else { renderizarTabelaFiltrada(lancamentosAtualizados); } } function duplicarLancamento(id) { let lancamentos = JSON.parse(localStorage.getItem('lancamentos')) || []; const lancamentoOriginal = lancamentos.find(l => l.id === id); if (lancamentoOriginal) { const dataOriginal = new Date(lancamentoOriginal.data + 'T00:00:00'); // Evita problemas de fuso horário local const proximoMes = new Date(dataOriginal.getFullYear(), dataOriginal.getMonth() + 1, dataOriginal.getDate()); const novoLancamentoDuplicado = { ...lancamentoOriginal, id: Date.now() + Math.random(), data: proximoMes.toISOString().split('T')[0], status: 'pendente' }; lancamentos.push(novoLancamentoDuplicado); localStorage.setItem('lancamentos', JSON.stringify(lancamentos)); const inputCompetencia = document.getElementById('filtroCompetencia'); if (inputCompetencia && inputCompetencia.value) { filtrarLancamentosPorCompetencia(inputCompetencia.value); } else { renderizarTabelaFiltrada(lancamentos); } alert('Lançamento duplicado para o próximo mês com status Pendente!'); } else { alert('Lançamento não encontrado para duplicação.'); } } // ========================================== // 6. FILTRO DE COMPETÊNCIA MENSAL // ========================================== function inicializarFiltroCompetencia() { const inputCompetencia = document.getElementById('filtroCompetencia'); if (inputCompetencia) { const hoje = new Date(); const competenciaAtual = `${hoje.getFullYear()}-${(hoje.getMonth() + 1).toString().padStart(2, '0')}`; inputCompetencia.value = competenciaAtual; inputCompetencia.addEventListener('change', function() { filtrarLancamentosPorCompetencia(this.value); }); filtrarLancamentosPorCompetencia(competenciaAtual); } else { renderizarTabelaFiltrada(JSON.parse(localStorage.getItem('lancamentos')) || []); } } function filtrarLancamentosPorCompetencia(competencia) { let todosLancamentos = JSON.parse(localStorage.getItem('lancamentos')) || []; if (!competencia) { renderizarTabelaFiltrada(todosLancamentos); return; } const lancamentosFiltrados = todosLancamentos.filter(lancamento => { const dataLancamento = new Date(lancamento.data + 'T00:00:00'); const anoLancamento = dataLancamento.getFullYear(); const mesLancamento = (dataLancamento.getMonth() + 1).toString().padStart(2, '0'); return `${anoLancamento}-${mesLancamento}` === competencia; }); renderizarTabelaFiltrada(lancamentosFiltrados); } // ========================================== // 7. SISTEMA DE ABAS LATERAIS // ========================================== function inicializarSistemaAbas() { const abas = document.querySelectorAll('.aba'); const conteudosAba = document.querySelectorAll('.conteudo-aba'); abas.forEach(aba => { aba.addEventListener('click', () => { abas.forEach(a => a.classList.remove('ativa')); aba.classList.add('ativa'); conteudosAba.forEach(c => c.classList.remove('ativa')); const idConteudo = aba.getAttribute('data-aba'); const alvo = document.getElementById(idConteudo); if (alvo) alvo.classList.add('ativa'); }); }); if (abas.length > 0) abas[0].click(); } // ========================================== // 8. EXPORTAÇÃO DE RELATÓRIOS (PDF) // ========================================== async function exportarParaPDF() { const competenciaSelecionada = document.getElementById('filtroCompetencia')?.value; let lancamentosParaExportar = JSON.parse(localStorage.getItem('lancamentos')) || []; if (competenciaSelecionada) { lancamentosParaExportar = lancamentosParaExportar.filter(lancamento => { const dataLancamento = new Date(lancamento.data + 'T00:00:00'); const anoLancamento = dataLancamento.getFullYear(); const mesLancamento = (dataLancamento.getMonth() + 1).toString().padStart(2, '0'); return `${anoLancamento}-${mesLancamento}` === competenciaSelecionada; }); } else { console.warn("Nenhuma competência selecionada. Exportando todos os registros."); } const { jsPDF } = window.jspdf || {}; if (!jsPDF) { alert("Erro: Biblioteca jsPDF não encontrada. Certifique-se de adicioná-la no seu HTML."); return; } const doc = new jsPDF(); doc.setFontSize(18); doc.text("Relatório Financeiro", 10, 10); doc.setFontSize(12); doc.text(competenciaSelecionada ? `Competência: ${competenciaSelecionada}` : "Período: Todos os Registros", 10, 20); let yPosition = 35; doc.setFontSize(10); doc.text("Data | Descrição | Categoria | Tipo | Valor", 10, yPosition); doc.line(10, yPosition + 2, 200, yPosition + 2); yPosition += 10; lancamentosParaExportar.forEach(l => { if (yPosition > 280) { // Cria nova página se estourar o limite vertical doc.addPage(); yPosition = 20; } const dataFormatada = new Date(l.data + 'T00:00:00').toLocaleDateString('pt-BR'); const valorFormatado = formatarMoeda(l.valor); const tipoLimpo = l.tipo.replace('_', ' '); doc.text(`${dataFormatada} | ${l.descricao} | ${l.categoria || '-'} | ${tipoLimpo} | ${valorFormatado}`, 10, yPosition); yPosition += 8; }); doc.save(`Relatorio_Financeiro_${competenciaSelecionada || 'Geral'}.pdf`); }
